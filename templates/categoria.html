@app.route("/categoria/") def categoria(categoria): carpeta = os.path.join( "static", "thumbnails", categoria ) videos = [] if os.path.exists(carpeta): for archivo in os.listdir(carpeta): if archivo.lower().endswith(".jpg"): nombre = os.path.splitext( archivo )[0] videos.append({ "nombre": nombre, "thumb": archivo }) videos.sort( key=lambda x: x["nombre"] ) return render_template( "categoria.html", categoria=categoria, videos=videos )
