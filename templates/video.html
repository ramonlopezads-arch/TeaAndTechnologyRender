@app.route("/video//") def video(categoria, nombre): carpeta = os.path.join( "static", "thumbnails", categoria ) videos = [] for archivo in os.listdir(carpeta): if archivo.lower().endswith(".jpg"): videos.append( os.path.splitext( archivo )[0] ) videos.sort() indice = videos.index(nombre) anterior = videos[ (indice - 1) % len(videos) ] siguiente = videos[ (indice + 1) % len(videos) ] ruta_video = ( f"https://res.cloudinary.com/" f"mh5fkptg/video/upload/" f"{nombre}.mp4" ) return render_template( "video.html", categoria=categoria, ruta_video=ruta_video, anterior=anterior, siguiente=siguiente )
